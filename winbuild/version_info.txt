# Windows executable metadata, read by PyInstaller's `version=` argument.
#
# This is what Explorer shows on the Details tab and what SmartScreen quotes
# when it warns about an unsigned binary. Filling it in properly does not make
# the warning go away - only a code-signing certificate does that - but an
# executable with a real company, product and description is a great deal
# easier for someone to trust than one with blank fields.
#
# Regenerate with: python winbuild/make_version_info.py

VSVersionInfo(
  ffi=FixedFileInfo(
    filevers=(0, 2, 2, 0),
    prodvers=(0, 2, 2, 0),
    mask=0x3f,
    flags=0x0,
    OS=0x40004,
    fileType=0x1,
    subtype=0x0,
    date=(0, 0)
  ),
  kids=[
    StringFileInfo(
      [
        StringTable(
          '040904B0',
          [StringStruct('CompanyName', 'at0m-b0mb'),
           StringStruct('FileDescription', 'Lares - autonomous Windows hardening agent'),
           StringStruct('FileVersion', '0.2.2'),
           StringStruct('InternalName', 'lares'),
           StringStruct('LegalCopyright', 'MIT licensed. See LICENSE.'),
           StringStruct('OriginalFilename', 'lares.exe'),
           StringStruct('ProductName', 'Lares'),
           StringStruct('ProductVersion', '0.2.2')]
        )
      ]
    ),
    VarFileInfo([VarStruct('Translation', [1033, 1200])])
  ]
)
